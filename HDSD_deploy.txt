3. NỘI DUNG deploy.sh (đã tối ưu cho Mongo Atlas + PM2 + Nginx)

#!/bin/bash
set -e

# ==============================
# Config
# ==============================
APP_NAME="tintuc"
APP_PORT=8069
APP_DIR="/var/www/$APP_NAME"

echo "===== UPDATE VPS ====="
sudo apt update -y && sudo apt upgrade -y

echo "===== INSTALL DEPENDENCIES ====="
sudo apt install -y curl git nginx ufw

echo "===== INSTALL NODEJS + PM2 ====="
if ! command -v node >/dev/null 2>&1; then
  curl -fsSL https://deb.nodesource.com/setup_20.x | sudo -E bash -
  sudo apt install -y nodejs build-essential
fi
sudo npm install -g pm2

echo "===== CLONE OR PULL GIT ====="
if [ -d "$APP_DIR/.git" ]; then
  cd "$APP_DIR"
  git pull origin main
else

  sudo chown $USER:$USER "$APP_DIR"
  git clone "$REPO_URL" "$APP_DIR"
  cd "$APP_DIR"
fi

echo "===== INSTALL NODE MODULES ====="
npm install

echo "===== CREATE .ENV (Mongo Atlas) ====="

echo "===== START WITH PM2 ====="
if [ -f ecosystem.config.js ]; then
  pm2 start ecosystem.config.js --env production || pm2 restart "$APP_NAME"
else
  # fallback nếu repo không có ecosystem.config.js
  pm2 start src/server.js --name "$APP_NAME"
fi
pm2 save
pm2 startup systemd -u $USER --hp $HOME >/dev/null 2>&1 || true

echo "===== CONFIGURE NGINX ====="
NGINX_CONF="/etc/nginx/sites-available/$APP_NAME"
sudo tee $NGINX_CONF > /dev/null <<EOF
server {
    listen 80;
    listen [::]:80;
    server_name $DOMAIN;

    location / {
        proxy_pass http://localhost:$APP_PORT;
        proxy_http_version 1.1;
        proxy_read_timeout 36000s;

        proxy_set_header Host \$host;
        proxy_set_header X-Real-IP \$remote_addr;
        proxy_set_header X-Forwarded-For \$proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto \$scheme;

4. Chạy script
chmod +x deploy.sh
./deploy.sh

Kiểm tra:

pm2 list
pm2 logs --lines 100

Nếu cần sửa ENV sau này:

cd /var/www/tintuc
nano .env
pm2 restart tintuc
----------------
5. Trỏ domain về VPS (làm trên inet như bthg)
6. Cấp SSL (HTTPS) cho domain (khuyên dùng)
----------
sudo apt install -y certbot python3-certbot-nginx
sudo certbot --nginx -d backend.dantri24h.com
# nhập email, đồng ý điều khoản, chọn redirect -> xong
----------
7. Lệnh cập nhật nhanh lần sau (pull code + restart)
----------
echo "alias deploy-tintuc='cd /var/www/tintuc && git pull origin main && npm install && pm2 restart tintuc && pm2 save'" >> ~/.bashrc
source ~/.bashrc
# Từ đây, mỗi lần bạn push code mới lên GitHub, vào VPS chỉ cần:
deploy-tintuc

👉 Làm vậy thì bất kỳ IP nào cũng SSH được, không còn phụ thuộc vào IP mạng của bạn nữa.
ufw allow 22/tcp
ufw reload
